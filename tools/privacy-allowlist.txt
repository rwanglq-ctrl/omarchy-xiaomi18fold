# privacy-allowlist.txt — regex per line, matched against file:line:rule
# output from the scanner. Lines starting with # are comments; blank lines ignored.
#
# Each entry is a Perl-compatible regex (grep -P / rg -P).
# Be specific: match against the full "file:line:rule" string.

# Git SHA hashes in patch headers and BASE file are not secrets
patches/BASE:.*:base64-blob
patches/BASE:.*:hex-blob
patches/.*:1:base64-blob
patches/.*:1:hex-blob

# Diffstat lines in patches (contain hex-like strings from filenames)
patches/.*:.*:base64-blob
patches/.*:.*:hex-blob

# noreply@ and example.com/example.org emails are allowed globally (handled in scanner logic)
# These don't need entries here — the scanner skips them before reporting.

# The privacy scanner's own code references rule names that may contain owner terms
privacy-scan\.sh:.*:private-pattern
privacy-allowlist\.txt:.*:private-pattern
privacy-selftest\.sh:.*:private-pattern

# Public SHA-256 checksums for vendored upstream objects
vendor/SHA256SUMS:[1-4]:base64-blob
vendor/SHA256SUMS:[1-4]:hex-blob

# Public OpenPGP ALARM build key, fingerprint checked during signature verification
vendor/alarm-build-key\.asc:.*:base64-blob

# Public upstream commit and signing-key fingerprint
vendor/README\.md:(3|25):(base64-blob|hex-blob)

# Example GitHub release URL is a scanner false positive
docs/china-network\.md:108:base64-blob
